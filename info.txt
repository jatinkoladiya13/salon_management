#This is run in shell after create template schema 
    >>> from tenant.models import Tenant, Domain
    >>> t = Tenant.objects.create(
    ...     name="Template Schema",
    ...     schema_name="template_schema",
    ...     subscription_status="active"
    ... )
    >>> Domain.objects.create(
    ...     domain="template.localhost",
    ...     tenant=t,
    ...     is_primary=True
    ... )
    > from django.db import connection, transaction
    >>> cursor = connection.cursor()
    >>> cursor.execute('CREATE SCHEMA IF NOT EXISTS template_schema;')
    >>> transaction.commit()
    >>> print("Done!")

    #run 
    python manage.py migrate_schemas --schema=template_schema
    
    now compelete the template schema
